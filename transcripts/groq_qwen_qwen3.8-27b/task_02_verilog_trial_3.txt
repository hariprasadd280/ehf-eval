Write a synthesizable Verilog module named 'clk_divider' with ports: input wire clk, input wire rst, output reg clk_out. The module divides clk by 4 using asynchronous active-high reset. Return ONLY raw Verilog code inside a markdown verilog code block.

module clk_divider (
    input wire clk,
    input wire rst,
    output reg clk_out
);

    reg [1:0] counter;

    always @(posedge clk or posedge rst) begin
        if (rst) begin
            counter <= 2'd0;
            clk_out <= 1'b0;
        end else begin
            counter <= counter + 1'b1;
            clk_out <= counter[1];
        end
    end

endmodule
